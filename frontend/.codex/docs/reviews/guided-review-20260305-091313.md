export class MonitoringListAuthorizationService {
  private readonly currentSessionService = inject(CurrentSessionService)
  private readonly monitoringListsDataService = inject(
    MonitoringListsDataService,
  )
  private readonly subscriberRoleService = inject(SubscriberRoleService)

  public canEditMonitoringList$ =
    this.monitoringListsDataService.monitoringList$.pipe(
      combineLatestWith(this.subscriberRoleService.role$),
      map(
        ([monitoringList, role]) =>
          !!monitoringList && this.canEditMonitoringList(role),
      ),
    )

  canEditMonitoringList(
    role: SubscriptionRole | null = this.subscriberRoleService.currentRole,
  ): boolean {
    const requiredPermission =
      role === 'owner'
        ? 'write:owned:monitoring_lists'
        : 'write:org:monitoring_lists'
    return this.hasPermission(requiredPermission)
  }

## Checkpoint 5: Control-Flow Deep Dive
- Happy path: route loads -> subscriber role is loaded -> authorization service emits editability -> templates show/hide actions via directive/service booleans.
- Edge/error path (critical): user deletes own subscription without org-subscription permission -> destroy path mutates role -> ownership check runs with mutated role.
- Resilience semantics: no retry/idempotency concerns changed; key risk is local state ordering in async subscription callbacks.
- Domain invariant safety: conditional; "remove own access => leave protected view" invariant can fail.
- Security call: no backend auth bypass shown; this is UI authorization consistency and navigation correctness.

public destroyAndNotify(subscription: Subscription) {
  return super
    .destroy(subscription.id)
    .pipe(tap(() => this.notifyRoleMutation(subscription, null)))
}

private notifyRoleMutation(
  subscription: Subscription,
  role = subscription.role ?? null,
) {
  const currentMembershipId =
    this.currentSessionService.currentSession?.membershipId
  if (!currentMembershipId) return

  this.subscriberRoleService.notifySubscriptionMutated(
    subscription,
    currentMembershipId,
    role,
  )
}
